sudo bash <<'ROOT'
set -Eeuo pipefail

cd /opt/durion/alpha/backend

ENV_FILE=/opt/durion/alpha/.env
OVERRIDE_FILE=/opt/durion/alpha/docker-compose.prod.yml

compose() {
  docker compose \
    -f docker-compose.yml \
    -f "$OVERRIDE_FILE" \
    --env-file "$ENV_FILE" \
    "$@"
}

POSTGRES_USER="$(grep '^POSTGRES_USER=' "$ENV_FILE" | cut -d= -f2-)"

echo "Stopping stack..."
compose stop

echo "Finding Postgres data volume..."
POSTGRES_CONTAINER="$(compose ps -q postgres)"
if [[ -z "$POSTGRES_CONTAINER" ]]; then
  echo "Could not find postgres container" >&2
  exit 1
fi

POSTGRES_VOLUME="$(docker inspect "$POSTGRES_CONTAINER" --format '{{range .Mounts}}{{if eq .Destination "/var/lib/postgresql/data"}}{{.Name}}{{end}}{{end}}')"
if [[ -z "$POSTGRES_VOLUME" ]]; then
  echo "Could not determine Postgres data volume" >&2
  exit 1
fi

echo "Removing volume: $POSTGRES_VOLUME"
compose rm -sf postgres
docker volume rm "$POSTGRES_VOLUME"

echo "Recreating Postgres..."
compose up -d postgres

echo "Waiting for Postgres to become ready..."
until compose exec -T postgres pg_isready -U "$POSTGRES_USER" >/dev/null 2>&1; do
  sleep 2
done

echo "Databases after bootstrap:"
compose exec -T postgres psql -U "$POSTGRES_USER" -d postgres -c '\l'

echo "Starting rest of stack..."
compose up -d
compose ps
